import "jsr:@supabase/functions-js/edge-runtime.d.ts";

const BREVO_API_KEY = Deno.env.get('BREVO_API_KEY')
const SITE_URL = 'https://remontco.replit.app'

interface CompanyPayload {
  type: 'INSERT' | 'UPDATE' | 'DELETE'
  table: string
  record: {
    id: string
    name: string
    email: string
    eik: string
    city: string
    is_verified: boolean
    verified_at: string | null
    moderation_reason: string | null
    created_at: string
  }
  old_record?: {
    is_verified: boolean
  }
}

Deno.serve(async (req: Request) => {
  try {
    const payload: CompanyPayload = await req.json()
    
    if (payload.type !== 'UPDATE') {
      return new Response(JSON.stringify({ message: 'Not an update event' }), { 
        status: 200,
        headers: { 'Content-Type': 'application/json' }
      })
    }

    const company = payload.record
    const oldRecord = payload.old_record

    if (!oldRecord || oldRecord.is_verified === company.is_verified) {
      return new Response(JSON.stringify({ message: 'Verification status unchanged' }), { 
        status: 200,
        headers: { 'Content-Type': 'application/json' }
      })
    }

    const isApproved = company.is_verified === true
    const reason = company.moderation_reason

    let subject: string
    let headerText: string
    let headerColor: string
    let bodyContent: string

    if (isApproved) {
      subject = `Вашата фирма "${company.name}" беше одобрена!`
      headerText = 'Профилът Ви е Верифициран!'
      headerColor = 'linear-gradient(135deg, #059669, #047857)'
      bodyContent = `
        <p style="color: #374151; font-size: 16px; line-height: 1.6;">Здравейте,</p>
        <p style="color: #374151; font-size: 16px; line-height: 1.6;">
          Радваме се да Ви съобщим, че Вашата фирма <strong>"${company.name}"</strong> (ЕИК: ${company.eik}) беше успешно верифицирана.
        </p>
        <p style="color: #374151; font-size: 16px; line-height: 1.6;">Вече имате пълен достъп до всички функционалности на платформата.</p>
        <div style="margin-top: 30px; text-align: center;">
          <a href="${SITE_URL}/dashboard/company.html" 
             style="display: inline-block; background: #059669; color: white; padding: 14px 32px; 
                    border-radius: 8px; text-decoration: none; font-weight: 600;">
            Към Моето Табло
          </a>
        </div>
      `
    } else {
      subject = `Вашата фирма "${company.name}" не беше одобрена`
      headerText = 'Профилът Ви не беше одобрен'
      headerColor = 'linear-gradient(135deg, #DC2626, #B91C1C)'
      bodyContent = `
        <p style="color: #374151; font-size: 16px; line-height: 1.6;">Здравейте,</p>
        <p style="color: #374151; font-size: 16px; line-height: 1.6;">
          За съжаление, Вашата фирма <strong>"${company.name}"</strong> не беше одобрена.
        </p>
        ${reason ? `
        <div style="background: #FEF2F2; border-left: 4px solid #DC2626; padding: 15px 20px; margin: 20px 0;">
          <p style="color: #991B1B; margin: 0; font-weight: 600;">Причина:</p>
          <p style="color: #7F1D1D; margin: 10px 0 0 0;">${reason}</p>
        </div>
        ` : ''}
      `
    }

    const emailContent = {
      sender: { name: 'RemontCo', email: 'noreply@remont.co' },
      to: [{ email: company.email, name: company.name }],
      subject: subject,
      htmlContent: `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; padding: 20px;">
          <div style="background: ${headerColor}; padding: 30px; border-radius: 12px 12px 0 0; text-align: center;">
            <h1 style="color: white; margin: 0; font-size: 24px;">${headerText}</h1>
          </div>
          <div style="background: #f8fafc; padding: 30px; border: 1px solid #e2e8f0; border-top: none;">
            ${bodyContent}
          </div>
          <div style="background: #1e293b; padding: 20px; border-radius: 0 0 12px 12px; text-align: center;">
            <p style="color: #94a3b8; margin: 0; font-size: 14px;">RemontCo</p>
          </div>
        </div>
      `
    }

    const response = await fetch('https://api.brevo.com/v3/smtp/email', {
      method: 'POST',
      headers: {
        'Accept': 'application/json',
        'Content-Type': 'application/json',
        'api-key': BREVO_API_KEY!
      },
      body: JSON.stringify(emailContent)
    })

    if (!response.ok) {
      const error = await response.text()
      console.error('Brevo API error:', error)
      return new Response(JSON.stringify({ error }), { status: 500 })
    }

    return new Response(JSON.stringify({ success: true }), { 
      status: 200,
      headers: { 'Content-Type': 'application/json' }
    })

  } catch (error) {
    console.error('Error:', error)
    return new Response(JSON.stringify({ error: error.message }), { status: 500 })
  }
})
