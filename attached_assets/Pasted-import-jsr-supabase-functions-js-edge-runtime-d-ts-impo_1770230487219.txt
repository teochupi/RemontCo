import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from 'jsr:@supabase/supabase-js@2';

const BREVO_API_KEY = "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";
const SENDER_EMAIL = "chupetlov.teodor@gmail.com";
const APP_URL = "http://localhost:3000"; // TODO: Change for production

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { quote_id, status } = await req.json();
    const serviceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');

    if (!quote_id || !status) {
        return new Response(JSON.stringify({ error: "Missing quote_id or status" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    const supabaseClient = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      serviceKey ?? ''
    );

    // 1. Get Quote & Company info
    const { data: quote, error: quoteError } = await supabaseClient
      .from('quotes')
      .select('id, job_id, company_id, price, message')
      .eq('id', quote_id)
      .single();

    if (quoteError || !quote) {
        return new Response(JSON.stringify({ error: "Quote not found" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    // 2. Get Job Title & Consumer Name
    const { data: job, error: jobError } = await supabaseClient
      .from('jobs')
      .select('title, consumer_id')
      .eq('id', quote.job_id)
      .single();

    if (jobError) return new Response(JSON.stringify({ error: "Job fetch failed" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });

    const { data: consumer } = await supabaseClient
      .from('profiles')
      .select('first_name, last_name, email')
      .eq('id', job.consumer_id)
      .single();

    const consumerName = consumer ? `${consumer.first_name || ''} ${consumer.last_name || ''}`.trim() || 'Потребител' : 'Потребител';

    // 3. Get Company Owner to send email to
    const { data: company, error: companyError } = await supabaseClient
      .from('companies')
      .select('owner_id, name')
      .eq('id', quote.company_id)
      .single();

    if (companyError || !company) return new Response(JSON.stringify({ error: "Company not found" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });

    const { data: owner } = await supabaseClient
      .from('profiles')
      .select('email, first_name')
      .eq('id', company.owner_id)
      .single();

    if (!owner || !owner.email) {
        return new Response(JSON.stringify({ error: "Company owner has no email" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    // 4. Prepare Email Content
    const isAccepted = status === 'accepted';
    const actionText = isAccepted ? 'ПРЕГЛЕДАНА / ПОТВЪРДЕНА' : 'ОТКАЗАНА';
    const color = isAccepted ? '#198754' : '#dc3545'; // Green or Red

    const emailData = {
      sender: { name: "RemontCo", email: SENDER_EMAIL },
      to: [{ email: owner.email, name: owner.first_name || "Partner" }],
      subject: `Статус на оферта: ${actionText} от клиент`,
      htmlContent: `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; padding: 20px; border: 1px solid #eee; border-radius: 10px;">
            <h2 style="color: ${color}; text-align: center;">Офертата е ${actionText.toLowerCase()}</h2>
            <p>Здравейте, <strong>${company.name}</strong>,</p>
            <p>Клиентът <strong>${consumerName}</strong> току-що промени статуса на вашата оферта за обект "<strong>${job.title}</strong>".</p>
            
            <div style="background-color: #f8f9fa; padding: 15px; border-radius: 5px; margin: 20px 0; border-left: 5px solid ${color};">
                <p style="margin: 5px 0;"><strong>Нов Статус:</strong> <span style="color: ${color}; font-weight: bold;">${actionText}</span></p>
                <p style="margin: 5px 0;"><strong>Вашата цена:</strong> ${quote.price} BGN</p>
            </div>

            ${isAccepted ? `
                <p>Клиентът потвърди, че е получил и прегледал офертата ви. Може да очаквате връзка от него!</p>
                <div style="text-align: center; margin-top: 25px;">
                    <a href="${APP_URL}/dashboard/company.html#my-offers" style="background-color: #198754; color: white; padding: 12px 24px; text-decoration: none; border-radius: 25px; font-weight: bold;">Към моите оферти</a>
                </div>
            ` : `
                <p>За съжаление клиентът е отказал тази оферта.</p>
            `}
            
            <p style="font-size: 12px; color: #888; margin-top: 30px; text-align: center;">
                RemontCo Platform<br>
                Автоматично известие.
            </p>
        </div>
      `
    };

    // 5. Send
    const brevoResponse = await fetch('https://api.brevo.com/v3/smtp/email', {
      method: 'POST',
      headers: {
        'accept': 'application/json',
        'api-key': BREVO_API_KEY,
        'content-type': 'application/json'
      },
      body: JSON.stringify(emailData)
    });

    if (!brevoResponse.ok) {
        const txt = await brevoResponse.text();
        return new Response(JSON.stringify({ error: "Brevo Error", details: txt }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    return new Response(JSON.stringify({ success: true }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 200
    });

  } catch (err) {
    return new Response(JSON.stringify({ error: "Global Exception", details: err.message }), { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200
    });
  }
});
