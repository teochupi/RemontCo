import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from 'jsr:@supabase/supabase-js@2';

const BREVO_API_KEY = Deno.env.get('BREVO_API_KEY');
const SENDER_EMAIL = "chupetlov.teodor@gmail.com";
const APP_URL = "https://remontco.bg"; 

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { record } = await req.json();
    const serviceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');

    if (!record || !record.job_id) {
        return new Response(JSON.stringify({ error: "Missing data" }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    const supabaseClient = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      serviceKey ?? ''
    );

    // 1. Get Job
    const { data: job, error: jobError } = await supabaseClient
      .from('jobs')
      .select('consumer_id, title')
      .eq('id', record.job_id)
      .single();

    if (jobError || !job) {
        return new Response(JSON.stringify({ error: "Job fetch failed", details: jobError }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }
    
    // 2. Get Consumer
    const { data: consumer, error: consumerError } = await supabaseClient
      .from('profiles')
      .select('email, first_name')
      .eq('id', job.consumer_id)
      .single();

    if (consumerError || !consumer || !consumer.email) {
         return new Response(JSON.stringify({ error: "Consumer fetch failed", details: consumerError }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    // 3. Get Company Name
    let companyName = "Фирма";
    if (record.company_id) {
        const { data: company } = await supabaseClient
            .from('companies')
            .select('name')
            .eq('id', record.company_id)
            .single();
        if (company) companyName = company.name;
    }

    // Prepare content
    const timelineText = record.timeline_days ? `${record.timeline_days} дни` : 'По договаряне';
    const messageText = record.message || 'Няма допълнително съобщение';

    // 4. Send Email via Brevo
    const emailData = {
      sender: { name: "RemontCo", email: SENDER_EMAIL },
      to: [{ email: consumer.email, name: consumer.first_name || "User" }],
      subject: `Нова оферта за: ${job.title}`,
      htmlContent: `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; padding: 20px; border: 1px solid #eee; border-radius: 10px;">
            <h2 style="color: #0d6efd; text-align: center;">Имате нова оферта!</h2>
            <p>Здравейте, ${consumer.first_name || ''}</p>
            <p>Фирма <strong>${companyName}</strong> току-що изпрати оферта за вашия проект "<strong>${job.title}</strong>".</p>
            
            <div style="background-color: #f8f9fa; padding: 15px; border-radius: 5px; margin: 20px 0;">
                <p style="margin: 5px 0;"><strong>Цена:</strong> ${record.price} BGN</p>
                <p style="margin: 5px 0;"><strong>Срок:</strong> ${timelineText}</p>
                <p style="margin: 5px 0;"><strong>Съобщение:</strong> <i>${messageText}</i></p>
            </div>

            <div style="text-align: center; margin-top: 25px;">
                <a href="${APP_URL}/dashboard/consumer.html#offers" style="background-color: #0d6efd; color: white; padding: 12px 24px; text-decoration: none; border-radius: 25px; font-weight: bold;">Виж Офертата</a>
            </div>
            
            <p style="font-size: 12px; color: #888; margin-top: 30px; text-align: center;">
                RemontCo Platform<br>
                Това е автоматично съобщение.
            </p>
        </div>
      `
    };

    const brevoResponse = await fetch('https://api.brevo.com/v3/smtp/email', {
      method: 'POST',
      headers: {
        'accept': 'application/json',
        'api-key': BREVO_API_KEY!,
        'content-type': 'application/json'
      },
      body: JSON.stringify(emailData)
    });

    if (!brevoResponse.ok) {
        const txt = await brevoResponse.text();
        return new Response(JSON.stringify({ error: "Brevo Error", details: txt }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 });
    }

    return new Response(JSON.stringify({ success: true, message: "Email sent" }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 200
    });

  } catch (err) {
    return new Response(JSON.stringify({ error: "Global Exception", details: err.message }), { 
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200 
    });
  }
});
