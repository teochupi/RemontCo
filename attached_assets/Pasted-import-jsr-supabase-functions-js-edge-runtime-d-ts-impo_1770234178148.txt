import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2.39.3';

const BREVO_API_KEY = Deno.env.get('BREVO_API_KEY');
const SENDER_EMAIL = 'chupetlov.teodor@gmail.com';
const ADMIN_EMAIL = 'chupetlov.teodor@gmail.com';
const SITE_URL = 'https://remontco.vercel.app'; 

Deno.serve(async (req) => {
  try {
    const bodyJson = await req.json();
    const { type, table, record, old_record } = bodyJson;
    
    const supabaseUrl = Deno.env.get('SUPABASE_URL') ?? '';
    const supabaseServiceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? '';
    const supabase = createClient(supabaseUrl, supabaseServiceKey);

    let recipientEmail = '';
    let subject = '';
    let htmlContent = '';

    // 1. New Message Notification
    if (table === 'messages' && type === 'INSERT') {
      const { data: recipient } = await supabase.from('profiles').select('email, username').eq('id', record.recipient_id).single();
      const { data: sender } = await supabase.from('profiles').select('username, first_name').eq('id', record.sender_id).single();
      
      if (recipient?.email) {
        recipientEmail = recipient.email;
        subject = 'Имате ново съобщение в RemontCo';
        const senderName = sender?.username || sender?.first_name || 'Потребител';
        htmlContent = `
            <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; color: #333; border: 1px solid #eee; border-radius: 10px; padding: 20px;">
                <h2 style="color: #0d6efd;">Ново съобщение</h2>
                <p>Здравейте, <strong>${recipient.username || ''}</strong>,</p>
                <p>Получихте ново съобщение от <strong>${senderName}</strong>:</p>
                <div style="background: #f8f9fa; padding: 15px; border-left: 4px solid #0d6efd; margin: 20px 0;">
                    ${(record.body || '').replace(/\n/g, '<br>')}
                </div>
                <div style="text-align: center;">
                    <a href="${SITE_URL}/dashboard/consumer.html" style="background: #0d6efd; color: white; padding: 12px 25px; text-decoration: none; border-radius: 5px; font-weight: bold; display: inline-block;">Виж в RemontCo</a>
                </div>
            </div>`;
      }
    }

    // 2. Quote (Offer) Notification
    if (table === 'quotes' && type === 'INSERT') {
      const { data: jobData } = await supabase
        .from('jobs')
        .select(`
            title,
            consumer_id,
            profiles:consumer_id (email, username)
        `)
        .eq('id', record.job_id)
        .single();
      
      const { data: company } = await supabase
        .from('companies')
        .select('name')
        .eq('id', record.company_id)
        .single();

      const profile = jobData?.profiles;
      if (profile?.email) {
        recipientEmail = profile.email;
        subject = `Нова оферта за "${jobData.title}" - RemontCo`;
        htmlContent = `
            <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; color: #333; padding: 20px; border: 1px solid #eee; border-radius: 10px;">
                <h2 style="color: #198754;">Получихте нова оферта!</h2>
                <p>Здравейте, <strong>${profile.username || ''}</strong>,</p>
                <p>Фирма <strong>${company?.name || 'Строителна фирма'}</strong> изпрати оферта за Вашата обява: <strong>"${jobData.title}"</strong>.</p>
                <div style="background: #e9ecef; padding: 15px; border-radius: 5px; margin: 20px 0;">
                    <p style="margin: 0;"><strong>Съобщение:</strong> ${record.message || 'Няма допълнително съобщение.'}</p>
                    <p style="margin: 5px 0 0 0;"><strong>Цена:</strong> ${record.price ? record.price + ' лв.' : 'По договаряне'}</p>
                </div>
                <div style="text-align: center;">
                    <a href="${SITE_URL}/dashboard/consumer.html" style="background: #198754; color: white; padding: 12px 25px; text-decoration: none; border-radius: 5px; font-weight: bold; display: inline-block;">Виж офертата</a>
                </div>
            </div>`;
      }
    }

    // 3. Status Change: Notify Consumer (Job Rejected/Needs Revision)
    if (table === 'jobs' && type === 'UPDATE' && record.status === 'rejected' && old_record?.status !== 'rejected') {
        const { data: consumer } = await supabase.from('profiles').select('email, username').eq('id', record.consumer_id).single();
        if (consumer?.email) {
            recipientEmail = consumer.email;
            subject = 'Вашата обява изисква корекция - RemontCo';
            htmlContent = `
                <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; color: #333; border: 1px solid #eee; border-radius: 10px; padding: 20px;">
                    <h2 style="color: #dc3545;">Обявата изисква корекция</h2>
                    <p>Здравейте, <strong>${consumer.username || ''}</strong>,</p>
                    <p>Вашата обява <strong>"${record.title}"</strong> е върната за преработка от администратор.</p>
                    <div style="background: #fff3f3; padding: 15px; border-left: 4px solid #dc3545; margin: 20px 0;">
                        <strong>Причина за връщане:</strong><br>
                        ${record.moderation_reason || 'Липсва конкретно описание.'}
                    </div>
                    <p>Моля, редактирайте обявата и я изпратете отново за одобрение.</p>
                    <div style="text-align: center;">
                        <a href="${SITE_URL}/dashboard/consumer.html" style="background: #dc3545; color: white; padding: 12px 25px; text-decoration: none; border-radius: 5px; font-weight: bold; display: inline-block;">Редактирай в RemontCo</a>
                    </div>
                </div>`;
        }
    }

    // 4. Admin: Job Notifications (New or Status to Pending)
    if (table === 'jobs' && (type === 'INSERT' || (type === 'UPDATE' && record.status === 'pending' && old_record?.status !== 'pending'))) {
        recipientEmail = ADMIN_EMAIL;
        subject = 'Нова обява за одобрение - RemontCo';
        htmlContent = `
            <div style="font-family: Arial, sans-serif; padding: 20px; border: 1px solid #eee; border-radius: 10px;">
                <h2 style="color: #0d6efd;">Нова обява чака одобрение</h2>
                <p><strong>Заглавие:</strong> ${record.title}</p>
                <p><strong>Описание:</strong> ${record.description}</p>
                <div style="margin-top:20px; text-align: center;">
                    <a href="${SITE_URL}/dashboard/admin.html" style="background: #0d6efd; color: white; padding: 10px 20px; text-decoration: none; border-radius: 5px; display: inline-block;">Към Админ Панела</a>
                </div>
            </div>`;
    }

    // 5. Admin: Company Registration Notifications
    if (table === 'companies' && (type === 'INSERT' || (type === 'UPDATE' && record.status === 'pending' && old_record?.status !== 'pending'))) {
        recipientEmail = ADMIN_EMAIL;
        subject = 'Нова фирма за верификация - RemontCo';
        htmlContent = `
            <div style="font-family: Arial, sans-serif; padding: 20px; border: 1px solid #eee; border-radius: 10px;">
                <h2 style="color: #0d6efd;">Нова фирма чака верификация</h2>
                <p><strong>Име:</strong> ${record.name}</p>
                <p><strong>ЕИК:</strong> ${record.eik}</p>
                <p><strong>Град:</strong> ${record.city}</p>
                <div style="margin-top:20px; text-align: center;">
                    <a href="${SITE_URL}/dashboard/admin.html" style="background: #0d6efd; color: white; padding: 10px 20px; text-decoration: none; border-radius: 5px; display: inline-block;">Към Админ Панела</a>
                </div>
            </div>`;
    }

    if (recipientEmail && htmlContent) {
        const res = await fetch('https://api.brevo.com/v3/smtp/email', {
            method: 'POST',
            headers: {
                'accept': 'application/json',
                'api-key': BREVO_API_KEY!,
                'content-type': 'application/json'
            },
            body: JSON.stringify({
                sender: { name: "RemontCo", email: SENDER_EMAIL },
                to: [{ email: recipientEmail }],
                subject: subject,
                htmlContent: htmlContent
            })
        });

        const resData = await res.json();
        return new Response(JSON.stringify(resData), { status: res.status });
    }

    return new Response(JSON.stringify({ message: 'No action needed' }), { status: 200 });

  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), { status: 500 });
  }
});
